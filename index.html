function drawRunwayBlueprint2D(airport) { const canvas = document.getElementById('blueprint-canvas'); const container = document.getElementById('blueprint-container'); canvas.width = container.clientWidth; canvas.height = container.clientHeight; const ctx = canvas.getContext('2d'); if (!ctx) return; const cx = canvas.width / 2; const cy = canvas.height / 2; ctx.clearRect(0, 0, canvas.width, canvas.height); // 1. Vẽ vòng tròn Radar & Trục la bàn ctx.strokeStyle = 'rgba(0, 243, 255, 0.15)'; ctx.lineWidth = 1; [50, 95, 140].forEach(r => { ctx.beginPath(); ctx.arc(cx, cy, r, 0, Math.PI * 2); ctx.stroke(); }); ctx.beginPath(); ctx.moveTo(cx, cy - 150); ctx.lineTo(cx, cy + 150); ctx.moveTo(cx - 150, cy); ctx.lineTo(cx + 150, cy); ctx.stroke(); ctx.fillStyle = '#00f3ff'; ctx.font = '10px "Share Tech Mono"'; ctx.fillText('N (000°)', cx - 18, cy - 152); ctx.fillText('S (180°)', cx - 18, cy + 158); ctx.fillText('E', cx + 152, cy + 4); ctx.fillText('W', cx - 162, cy + 4); // 2. KHỚP KÉP NÂNG CAO (DUAL-KEY MATCHING): // So khớp cả airport_ident, airport_ref (ID số), và mã IATA/ICAO, đồng thời dọn sạch khoảng trắng const targetIdent = cleanStr(airport.ident); const targetId = airport.id ? String(airport.id).trim() : null; const targetIata = cleanStr(airport.iata_code); const rawRws = runwaysList.filter(r => { const rwIdent = cleanStr(r.airport_ident); const rwRef = r.airport_ref ? String(r.airport_ref).trim() : null; return (targetIdent && rwIdent === targetIdent) || (targetId && rwRef === targetId) || (targetIata && rwIdent === targetIata); }); document.getElementById('airport-rws-count').innerText = `${rawRws.length} Runway(s)`; if (rawRws.length === 0) { ctx.fillStyle = '#888'; ctx.font = '12px "Share Tech Mono"'; ctx.textAlign = 'center'; ctx.fillText('> NO RUNWAY GEOMETRY DATA AVAILABLE', cx, cy); return; } // 3. Chuẩn hóa góc hướng (Heading) và số liệu const parsedRunways = rawRws.map(r => { let heading = r.heading_degT; if (heading === null || heading === undefined || isNaN(heading)) { const numMatch = (r.le_ident || '').match(/\d+/); heading = numMatch ? parseInt(numMatch[0], 10) * 10 : 0; } return { ...r, computedHeading: Number(heading) || 0, baseAngle: Math.round((Number(heading) || 0) % 180) }; }); // Kiểm tra tính đầy đủ của cặp tọa độ đầu/cuối const validCoordRunways = parsedRunways.filter(r => r.le_latitude_deg !== null && !isNaN(r.le_latitude_deg) && r.he_latitude_deg !== null && !isNaN(r.he_latitude_deg) ); if (validCoordRunways.length === parsedRunways.length && validCoordRunways.length > 0) { // --- PHƯƠNG ÁN 1: CÓ ĐỦ TỌA ĐỘ TRẮC ĐỊA 2 ĐẦU ĐƯỜNG BĂNG --- const refLat = airport.latitude_deg * (Math.PI / 180); const METERS_PER_DEG_LAT = 111320; const METERS_PER_DEG_LON = 111320 * Math.cos(refLat); let minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity; const projected = parsedRunways.map(r => { const leX = (r.le_longitude_deg - airport.longitude_deg) * METERS_PER_DEG_LON; const leY = (r.le_latitude_deg - airport.latitude_deg) * METERS_PER_DEG_LAT; const heX = (r.he_longitude_deg - airport.longitude_deg) * METERS_PER_DEG_LON; const heY = (r.he_latitude_deg - airport.latitude_deg) * METERS_PER_DEG_LAT; minX = Math.min(minX, leX, heX); maxX = Math.max(maxX, leX, heX); minY = Math.min(minY, leY, heY); maxY = Math.max(maxY, leY, heY); return { ...r, leX, leY, heX, heY }; }); const spanX = Math.max(maxX - minX, 800); const spanY = Math.max(maxY - minY, 800); const scale = 220 / Math.max(spanX, spanY); const midX = (minX + maxX) / 2; const midY = (minY + maxY) / 2; projected.forEach(r => { const p1x = cx + (r.leX - midX) * scale; const p1y = cy - (r.leY - midY) * scale; const p2x = cx + (r.heX - midX) * scale; const p2y = cy - (r.heY - midY) * scale; drawRunwayBetweenPoints(ctx, p1x, p1y, p2x, p2y, r.le_ident, r.he_ident); }); } else { // --- PHƯƠNG ÁN 2: THIẾU TỌA ĐỘ HOẶC TỌA ĐỘ LỖI (VẼ BẰNG CHIỀU DÀI + GÓC HƯỚNG THỰC TẾ) --- const groups = {}; parsedRunways.forEach(r => { if (!groups[r.baseAngle]) groups[r.baseAngle] = []; groups[r.baseAngle].push(r); }); Object.values(groups).forEach(group => { const total = group.length; group.forEach((r, idx) => { const lengthPx = Math.min(Math.max((Number(r.length_ft) || 6500) / 45, 120), 240); const widthPx = Math.max((Number(r.width_ft) || 150) / 14, 8); const rad = (r.computedHeading * Math.PI) / 180; let offset = 0; const ident = (r.le_ident || '').toUpperCase(); if (ident.includes('L')) offset = -35; else if (ident.includes('R')) offset = 35; else if (ident.includes('C')) offset = 0; else if (total > 1) { offset = (idx - (total - 1) / 2) * 40; } const offX = offset * Math.cos(rad); const offY = offset * Math.sin(rad); ctx.save(); ctx.translate(cx + offX, cy + offY); ctx.rotate(rad); drawRunwayBody(ctx, widthPx, lengthPx, r.le_ident, r.he_ident); ctx.restore(); }); }); } }
