// HÀM VẼ RUNWAY BLUEPRINT NÂNG CẤP: KHẮC PHỤC TRIỆT ĐỂ LỖI KHÔNG TÌM THẤY ĐƯỜNG BĂNG function drawRunwayBlueprint2D(airport) { const container = document.getElementById('blueprint-container'); const canvas = document.getElementById('blueprint-canvas'); // Đảm bảo lấy đúng kích thước thật kể cả khi container đang animate const w = container.clientWidth || 360; const h = container.clientHeight || 320; canvas.width = w; canvas.height = h; const ctx = canvas.getContext('2d'); if (!ctx) return; const cx = w / 2; const cy = h / 2; ctx.clearRect(0, 0, w, h); // 1. Vẽ vòng tròn Radar tọa độ ctx.strokeStyle = 'rgba(0, 243, 255, 0.15)'; ctx.lineWidth = 1; [50, 95, 140].forEach(r => { ctx.beginPath(); ctx.arc(cx, cy, r, 0, Math.PI * 2); ctx.stroke(); }); ctx.beginPath(); ctx.moveTo(cx, cy - 150); ctx.lineTo(cx, cy + 150); ctx.moveTo(cx - 150, cy); ctx.lineTo(cx + 150, cy); ctx.stroke(); ctx.fillStyle = '#00f3ff'; ctx.font = '10px "Share Tech Mono"'; ctx.fillText('N (000°)', cx - 18, cy - 152); ctx.fillText('S (180°)', cx - 18, cy + 158); ctx.fillText('E', cx + 152, cy + 4); ctx.fillText('W', cx - 162, cy + 4); // 2. TÌM KIẾM ĐA ĐIỀU KIỆN (Bắt cả ident, id, gps_code và iata_code để không bao giờ bị sót) const airportIdNum = Number(airport.id); const rawRws = runwaysList.filter(r => { if (!r) return false; const matchIdent = r.airport_ident && (r.airport_ident === airport.ident || r.airport_ident === airport.gps_code || r.airport_ident === airport.iata_code); const matchRef = r.airport_ref && Number(r.airport_ref) === airportIdNum; return matchIdent || matchRef; }); document.getElementById('airport-rws-count').innerText = `${rawRws.length} Runway(s)`; if (rawRws.length === 0) { ctx.fillStyle = '#ff7700'; ctx.font = '12px "Share Tech Mono"'; ctx.textAlign = 'center'; ctx.fillText('> NO RUNWAY GEOMETRY LINKED IN DATASET', cx, cy - 10); ctx.fillStyle = '#888'; ctx.font = '10px "Share Tech Mono"'; ctx.fillText(`(Searched IDENT: ${airport.ident} | REF: ${airport.id})`, cx, cy + 12); return; } // 3. Chuẩn hóa góc quay và dữ liệu số const parsedRunways = rawRws.map(r => { let heading = Number(r.heading_degT); if (isNaN(heading) || heading === null) { const numMatch = (r.le_ident || '').match(/\d+/); heading = numMatch ? parseInt(numMatch[0], 10) * 10 : 0; } return { ...r, length_ft: Number(r.length_ft) || 6500, width_ft: Number(r.width_ft) || 150, le_lat: Number(r.le_latitude_deg), le_lng: Number(r.le_longitude_deg), he_lat: Number(r.he_latitude_deg), he_lng: Number(r.he_longitude_deg), computedHeading: heading, baseAngle: Math.round(heading % 180) }; }); // 4. Kiểm tra tọa độ GPS thực const hasRealCoords = parsedRunways.some(r => !isNaN(r.le_lat) && !isNaN(r.he_lat) && r.le_lat !== 0 && r.he_lat !== 0); if (hasRealCoords && airport.latitude_deg) { const refLat = (Number(airport.latitude_deg) || 0) * (Math.PI / 180); const METERS_PER_DEG_LAT = 111320; const METERS_PER_DEG_LON = 111320 * Math.cos(refLat); let minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity; const projected = parsedRunways.map(r => { let leX = 0, leY = 0, heX = 0, heY = 0; if (!isNaN(r.le_lat) && !isNaN(r.he_lat) && r.le_lat !== 0) { leX = (r.le_lng - airport.longitude_deg) * METERS_PER_DEG_LON; leY = (r.le_lat - airport.latitude_deg) * METERS_PER_DEG_LAT; heX = (r.he_lng - airport.longitude_deg) * METERS_PER_DEG_LON; heY = (r.he_lat - airport.latitude_deg) * METERS_PER_DEG_LAT; } else { const len = r.length_ft * 0.3048; const rad = (r.computedHeading * Math.PI) / 180; leX = -(len / 2) * Math.sin(rad); leY = -(len / 2) * Math.cos(rad); heX = (len / 2) * Math.sin(rad); heY = (len / 2) * Math.cos(rad); } minX = Math.min(minX, leX, heX); maxX = Math.max(maxX, leX, heX); minY = Math.min(minY, leY, heY); maxY = Math.max(maxY, leY, heY); return { ...r, leX, leY, heX, heY }; }); const spanX = Math.max(maxX - minX, 800); const spanY = Math.max(maxY - minY, 800); const scale = 220 / Math.max(spanX, spanY); const midX = (minX + maxX) / 2; const midY = (minY + maxY) / 2; projected.forEach(r => { const p1x = cx + (r.leX - midX) * scale; const p1y = cy - (r.leY - midY) * scale; const p2x = cx + (r.heX - midX) * scale; const p2y = cy - (r.heY - midY) * scale; drawRunwayBetweenPoints(ctx, p1x, p1y, p2x, p2y, r.le_ident, r.he_ident); }); } else { // Thuật toán tách dải offset nếu thiếu tọa độ thực const groups = {}; parsedRunways.forEach(r => { if (!groups[r.baseAngle]) groups[r.baseAngle] = []; groups[r.baseAngle].push(r); }); Object.values(groups).forEach(group => { const total = group.length; group.forEach((r, idx) => { const lengthPx = Math.min(Math.max(r.length_ft / 45, 120), 240); const widthPx = Math.max(r.width_ft / 14, 8); const rad = (r.computedHeading * Math.PI) / 180; let offset = 0; const ident = (r.le_ident || '').toUpperCase(); if (ident.includes('L')) offset = -35; else if (ident.includes('R')) offset = 35; else if (ident.includes('C')) offset = 0; else if (total > 1) { offset = (idx - (total - 1) / 2) * 40; } const offX = offset * Math.cos(rad); const offY = offset * Math.sin(rad); ctx.save(); ctx.translate(cx + offX, cy + offY); ctx.rotate(rad); drawRunwayBody(ctx, widthPx, lengthPx, r.le_ident, r.he_ident); ctx.restore(); }); }); } }
